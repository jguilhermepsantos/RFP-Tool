import os
from datetime import datetime
from dotenv import load_dotenv
from supabase import create_client, Client
from pinecone import Pinecone
from openai import OpenAI

# Load environment variables
load_dotenv()
SUPABASE_URL = os.getenv("SUPABASE_URL")
SUPABASE_KEY = os.getenv("SUPABASE_KEY")
OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")
PINECONE_API_KEY = os.getenv("PINECONE_API_KEY")
PINECONE_ENV = os.getenv("PINECONE_ENV")  
PINECONE_INDEX_NAME = os.getenv("PINECONE_INDEX_NAME")  # e.g., "vtex-global"

# Initialize clients
supabase: Client = create_client(SUPABASE_URL, SUPABASE_KEY)
client = OpenAI(api_key=OPENAI_API_KEY)
# pinecone.init(api_key=PINECONE_API_KEY, environment=PINECONE_ENV)
# index = pinecone.Index(PINECONE_INDEX_NAME)
pc = Pinecone(api_key=os.getenv("PINECONE_API_KEY"))
index = pc.Index(os.getenv("PINECONE_INDEX_NAME"))

def embed_unprocessed_chunks():
    print("🔍 Fetching chunks where embedded = false...")
    response = supabase.table("chunks").select("*").eq("embedded", False).execute()
    chunks = response.data
    print(f"✅ Found {len(chunks)} unembedded chunks.")

    for chunk in chunks:
        chunk_id = chunk["id"]
        content = chunk["content"]
        document_id = chunk["document_id"]

        # Embed with OpenAI
        embedding = client.embeddings.create(
            model="text-embedding-3-small",
            input=[content]
        ).data[0].embedding

        # Upsert into Pinecone (no content storage here!)
        index.upsert(
            vectors=[
                (
                    chunk_id,  # Vector ID = same as chunk ID
                    embedding,
                    {
                        "scope": str(chunk.get("scope") or "global"),
                        "document_id": str(chunk.get("document_id") or "unknown")
                    }
                )
            ]
        )

        # Update Supabase to mark as embedded and set embedded_at timestamp
        supabase.table("chunks").update({
            "embedded": True,
            "embedded_at": datetime.utcnow().isoformat()
        }).eq("id", chunk_id).execute()

        print(f"✅ Embedded chunk {chunk_id} into Pinecone")

if __name__ == "__main__":
    embed_unprocessed_chunks()
